static const float digit_bitmaps[10] =
{
	480599.0, 139810.0, 476951.0, 476999.0, 350020.0, 464711.0, 464727.0, 476228.0, 481111.0, 481095.0
};

float print_value(float2 string_coords, float value, float integer_digits, float decimal_places)
{
	if (string_coords.y < 0.0 || string_coords.y >= 1.0)
	{
		return 0.0;
	}

	bool is_negative = value < 0.0;
	value = abs(value);

	float biggest_index = max(floor(log10(value)), 0.0);
	float digit_index = integer_digits - floor(string_coords.x);

	if (digit_index <= -decimal_places - 1.01)
	{
		return 0.0;
	}

	float character_bitmap = 0.0;

	if (digit_index > biggest_index)
	{
		if (is_negative && digit_index < biggest_index + 1.5)
		{
			character_bitmap = MINUS_BITMAP;
		}
	}
	else if (digit_index == -1.0)
	{
		character_bitmap = DECIMAL_POINT_BITMAP;
	}
	else
	{
		float reduced_value = value;

		if (digit_index < 0.0)
		{
			reduced_value = frac(value);
			digit_index += 1.0;
		}

		float digit_value = reduced_value / pow(10.0, digit_index);
		int digit = (int)floor(fmod(digit_value, 10.0));
		character_bitmap = digit_bitmaps[digit];
	}

	float bit_index = floor(frac(string_coords.x) * 4.0) + floor(string_coords.y * 5.0) * 4.0;

	return floor(fmod(character_bitmap / pow(2.0, bit_index), 2.0));
}
