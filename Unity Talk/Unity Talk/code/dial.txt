#define DEBUG_DIAL(uv, value, row, column) \
	{ \
		float4 debug_color; \
		if (debug_dial((uv), (value), (row), (column), debug_color)) \
		{ \
			return debug_color; \
		} \
	}

bool debug_dial(float2 uv, float value, int row, int column, out float4 color)
{
	float2 cell_uv;
	color = 0;

	if (!debug_cell(uv, row, 2, column, 2, cell_uv))
	{
		return false;
	}

	float radius_in_pixels = DIAL_RADIUS * _ScreenParams.y / ROW_COUNT;
	float2 position = (cell_uv * 2.0 - 1.0) / DIAL_RADIUS;
	float distance_from_center = length(position);

	if (distance_from_center > 1.0)
	{
		color = float4(BACKGROUND_COLOR, 1);
		return true;
	}

// HIGHLIGHT_START
	float pixel_turns = frac(atan2(position.y, position.x) / (2.0 * PI));
	float turn_difference = abs(frac(pixel_turns - value + 0.5) - 0.5);
	float distance_to_needle = turn_difference * 2.0 * PI * distance_from_center * radius_in_pixels;

	float needle_width = lerp(HUB_WIDTH, TIP_WIDTH, distance_from_center);
	bool is_on_needle = distance_to_needle < needle_width;
// HIGHLIGHT_END

	color = float4(is_on_needle ? FOREGROUND_COLOR : FACE_COLOR, 1);
	return true;
}

bool debug_cell(float2 uv, int row, int row_span, int column, int column_count, out float2 cell_uv)
{
	float row_size = _ScreenParams.y / ROW_COUNT;
	float2 pixel = float2(uv.x, 1.0 - uv.y) * _ScreenParams.xy;

	float cell_width = REGION_WIDTH_IN_ROWS * row_size / column_count;
	float2 cell_min = float2(column * cell_width, (row + 1) * row_size);
	float2 cell_size = float2(cell_width, row_span * row_size);

	float2 local = (pixel - cell_min) / cell_size;
	cell_uv = float2(local.x, 1.0 - local.y);

	return all(local >= 0.0) && all(local < 1.0);
}
