uint3 hash(uint3 value)
{
	value ^= value >> 16;
	value *= 0x7feb352du;
	value ^= value >> 15;
	value *= 0x846ca68bu;
	value ^= value >> 16;

	return value;
}

float4 Frag(v2f input) : SV_Target
{
// HIGHLIGHT_START
	uint color_index = _Id % _ColorCount;
	uint3 hashed = hash(color_index + uint3(0u, 0x68e31da4u, 0xb5297a4du));
	float3 color = float3(hashed >> 8) / 16777215.0;
// HIGHLIGHT_END

	return float4(color, 1);
}
