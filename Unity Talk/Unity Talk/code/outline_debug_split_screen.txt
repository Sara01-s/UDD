float4 Frag(v2f input) : SV_Target
{
	float2 uv = input.uv;
	float2 offset = _MainTex_TexelSize.xy * _JumpSize;

	float3 bottom_left  = tex2D(_MainTex, uv + offset * float2(-1, -1)).rgb;
	float3 bottom       = tex2D(_MainTex, uv + offset * float2( 0, -1)).rgb;
	float3 bottom_right = tex2D(_MainTex, uv + offset * float2( 1, -1)).rgb;
	float3 left         = tex2D(_MainTex, uv + offset * float2(-1,  0)).rgb;
	float3 right        = tex2D(_MainTex, uv + offset * float2( 1,  0)).rgb;
	float3 top_left     = tex2D(_MainTex, uv + offset * float2(-1,  1)).rgb;
	float3 top          = tex2D(_MainTex, uv + offset * float2( 0,  1)).rgb;
	float3 top_right    = tex2D(_MainTex, uv + offset * float2( 1,  1)).rgb;

	float3 gradient_x = (top_right + 2 * right + bottom_right) - (top_left + 2 * left + bottom_left);
	float3 gradient_y = (top_left + 2 * top + top_right) - (bottom_left + 2 * bottom + bottom_right);

	float edge = sqrt(dot(gradient_x, gradient_x) + dot(gradient_y, gradient_y)) * _EdgeFactor;

// HIGHLIGHT_START
#if TOOL_SHOW_EDGE
	if (uv.x < 0.5)
	{
		return float4(edge.xxx, 1);
	}
#endif
// HIGHLIGHT_END

	float4 color = tex2D(_MainTex, uv);

	return edge > _Threshold ? _LineColor : color;
}
