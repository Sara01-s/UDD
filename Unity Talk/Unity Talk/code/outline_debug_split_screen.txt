float4 Frag(v2f input) : SV_Target
{
	float2 uv = input.uv;
	float2 offset = _MainTex_TexelSize.xy * _JumpSize;

	float3 pixel00 = tex2D(_MainTex, uv + offset * float2(-1, -1)).rgb;
	float3 pixel10 = tex2D(_MainTex, uv + offset * float2( 0, -1)).rgb;
	//etc...
	
	float3 gradient_x = (pixel22 + 2 * pixel21 + pixel20) - (pixel02 + 2 * pixel01 + pixel00);
	float3 gradient_y = (pixel02 + 2 * pixel12 + pixel22) - (pixel00 + 2 * pixel10 + pixel20);

	float edge = sqrt(dot(gradient_x, gradient_x) + dot(gradient_y, gradient_y)) * _EdgeFactor;

// HIGHLIGHT_START
#if TOOL_SHOW_GRADIENT
	if (uv.x < 0.5)
	{
		return float4(edge.xxx, 1);
	}
#endif
// HIGHLIGHT_END

	float4 color = tex2D(_MainTex, uv);

	return edge > _Threshold ? _LineColor : color;
}
