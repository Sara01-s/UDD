bool debug_signed_bar(float2 uv, float value, int row, out float4 color)
{
	float2 cell_uv;
	color = 0;

	if (!debug_cell(uv, row, 1, 0, 1, cell_uv)) 
	{
		return false;
	}

	float position = cell_uv.x * 2.0 - 1.0;
	float clamped_value = clamp(value, -1.0, 1.0);

	bool is_on_bar = clamped_value >= 0.0
		? position >= 0.0 && position < clamped_value
		: position < 0.0 && position >= clamped_value;

	float3 bar_color = abs(value) <= 1.0 ? FOREGROUND_COLOR : OUT_OF_RANGE_COLOR;

	color = float4(is_on_bar ? bar_color : BACKGROUND_COLOR, 1);
	return true;
}
